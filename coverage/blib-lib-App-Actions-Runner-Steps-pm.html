=head1 DESCRIPTION

B<In short:> this module runs the steps of one job, one after another, in
the job's container, and passes values between them as GitHub does.  It is
a part of L<App::Actions::Runner>; you rarely use it on its own.

Executes a job's steps with C<docker exec> in its long-lived container, so
filesystem changes persist between steps while each step gets a fresh shell,
as on GitHub.  Each step gets its own C<GITHUB_ENV>, C<GITHUB_OUTPUT>,
C<GITHUB_PATH>, C<GITHUB_STEP_SUMMARY> and C<GITHUB_STATE> files, which are
read back after it: exported variables and C<PATH> additions reach the later
steps, outputs reach C<steps.ID.outputs>.

A step is a script (C<run>), a C<docker://> action, a local composite or
JavaScript action (C<uses: ./path>), or a well-known action that is a no-op
here.  Composite actions run their steps through the same machinery, one
level deeper.  JavaScript actions' C<pre> runs before C<main>, and C<post>
runs after the job's last step, in reverse order.

517# _run_post_actions
518# Purpose:     Run the post steps JavaScript actions deferred, in reverse
519#              order, as GitHub does, each under its own condition.
520# Entry:       None.
521# Exit status: Returns nothing.
522# Side effects: Runs node in the container; may mark the job failed.
523sub _run_post_actions {
524        my ($self) = @_;
525
526        my $ctx = $self->{ctx};
527        my $status = status_flags($self->{failed});
528
529        for my $post (reverse @{ $self->{post_actions} }) {
530                my %post_ctx = (
531                        %{$ctx},
532                        inputs => $post->{inputs},
533                        env => $self->{job_env},
534                        github => { %{ $ctx->{github} || {} }, action_path => $post->{action_path} },
535                );
536
537                my $should_run = eval { $self->{expression}->condition($post->{condition}, \%post_ctx, $status) };
538                if (!defined $should_run) {
539                        $self->_complain('runner.post.if_failed', { uses => $post->{uses}, error => trim_error($@) });
540                        next;
541                }
542                next unless $should_run;
543
544                $self->_say($self->i18n('runner.post.start', { uses => $post->{uses} }));
545
546                my $rc = eval {
547                        my $files = $self->_prepare_step_files(${ $self->{counter} } += 1, '-post');
548                        my %post_env = (%{ $self->{job_env} }, %{ $files->{vars} });
549                        $post_env{"STATE_$_"} = $post->{state}{$_} for keys %{ $post->{state} || {} };
550                        $self->_run_node_script($post->{script}, \%post_env, $post->{step_id}, $post->{action_path}, $post->{inputs});
551                };
552
553                # A failed post step fails the job, and later posts see failure().
554                if (!defined $rc) {
555                        $self->_complain('runner.post.failed', { uses => $post->{uses}, error => trim_error($@) });
556                } elsif ($rc != 0) {
557                        $self->{logger}->error($self->i18n('runner.post.exited', { uses => $post->{uses}, status => $rc }))
558                                unless $self->{quiet};
559                }
560                next if defined $rc && $rc == 0;
561                $self->{failed} = 1;
562                $status = status_flags(1);
563        }
564
565        return;
566}

568# _run_script_step
569# Purpose:     Run a `run:` step: write the script into the runner directory
570#              and execute it in the job container with the step's shell.
571# Entry:       $step; $step_number; $env - the step's environment;
572#              $context, $status - for interpolation.
573# Exit status: Returns a step result hash.
574# Side effects: Writes the script; runs docker exec; reads the step files.
575sub _run_script_step {
576        my ($self, $step, $step_number, $env, $context, $status) = @_;
577
578        my $shell = $self->_shell_for($step, $context, $status);
579        return $self->_result(outcome => 'failure', error => $self->i18n('runner.shell.missing', { shell => $shell->{name} }))
580                if $shell->{requires_bash} && !$self->{has_bash};
581
582        my $script = $self->{expression}->interpolate($step->{run}, $context, $status);
583        $script .= "\n" unless $script =~ /\n\z/;
584
585        my $files = $self->_prepare_step_files($step_number);
586        my $exec_env = $self->_exec_env($env, $files, defined $step->{id} ? $step->{id} : "__step_$step_number", $context->{github}{action_path});
587        my $workdir = $self->_working_directory($step, $context, $status);
588
589        # The script lives in the runner directory, which the container mounts.
590        my $script_name = "step-$step_number$shell->{extension}";
591        my $host_script = "$self->{temp_path}/scripts/$script_name";
592        write_container_file($host_script, $script, $self->i18n('runner.what.step_script', { file => $host_script }));
593        $self->_chmod_private($host_script);
594
595        my $command = $shell->{command};
596        my $quoted_script = _shell_quote("$RUNNER_DIR/scripts/$script_name");
597        $command =~ s/\{0\}/$quoted_script/g or $command .= " $quoted_script";
598
599        if ($self->{dry_run} && $self->{verbose}) {
600                print $self->i18n('runner.script.begin'), "\n", redact($self->{docker}, $script), $self->i18n('runner.script.end'), "\n";
601        }
602
603        my $rc = $self->{docker}->container_exec('-w', $workdir, @{ env_args($exec_env) }, $self->{container}, 'sh', '-c', $command);
604
605        return $self->_collected($files, $rc, 'runner.step.exited');
606}

608# _run_uses_step
609# Purpose:     Run a `uses:` step: a well-known no-op, a docker:// action or
610#              a local action; anything else is unsupported.
611# Entry:       As _run_script_step.
612# Exit status: Returns a step result hash.
613# Side effects: As the action it runs.
614sub _run_uses_step {
615        my ($self, $step, $step_number, $env, $context, $status) = @_;
616
617        my $action = $self->{expression}->interpolate($step->{uses}, $context, $status);
618        my $with = $step->{with} || {};
619
620        if ($action =~ $CHECKOUT) {
621                # The repository is mounted, so what checkout would fetch is ignored;
622                # each option that would have changed the checkout says so.
623                $self->_complain('runner.checkout.path') if defined $with->{path} && length $with->{path};
624                $self->_complain('runner.checkout.submodules') if defined $with->{submodules} && $with->{submodules} ne 'false';
625                $self->_complain('runner.checkout.ref') if defined $with->{ref} || defined $with->{repository};
626                return $self->_noop_result($self->i18n('runner.noop.checkout'));
627        }
628
629        if ($action =~ $SETUP_PERL) {
630                my $version = $with->{'perl-version'};
631                $version = $self->{expression}->interpolate($version, $context, $status) if defined $version;
632                return $self->_noop_result(defined $version
633                        ? $self->i18n('runner.noop.perl_version', { version => $version })
634                        : $self->i18n('runner.noop.perl'));
635        }
636
637        return $self->_noop_result($self->i18n('runner.noop.cache')) if $action =~ $NO_OP;
638        # The image is what is left of the resolved reference: it is not
639        # interpolated again, so a value from a step output is never read as an
640        # expression, and an empty one is reported as having no name.
641        return $self->_run_docker_action($step, $step_number, $1, $env, $context, $status) if $action =~ m{\Adocker://(.*)\z}s;
642        return $self->_run_local_action($step, $step_number, $action, $env, $context, $status) if $action =~ m{\A\.\.?/};
643
644        return $self->_unsupported_action($action, $self->i18n('runner.unsupported.remote'));
645}

647# _run_docker_action
648# Purpose:     Run a docker:// action in a container of its own on the job's
649#              network.  An image a step output decides is first known here,
650#              so it is checked and prepared here.
651# Entry:       $step; $step_number; $image - the resolved text after docker://;
652#              $env; $context; $status.
653# Exit status: Returns a step result hash.
654# Side effects: May pull the image; runs a container; reads the step files.
655sub _run_docker_action {
656        my ($self, $step, $step_number, $image, $env, $context, $status) = @_;
657
658        my $job_id = $self->{job}{base_id};
659
660        # It must be a real name before a container is started from it.
661        $self->throw('runner.docker_action.no_image', { uses => $step->{uses}, job => $job_id }) unless length $image;
662        $self->throw('runner.docker_action.bad_image', { uses => $step->{uses}, job => $job_id, image => $image })
663                if $image =~ /\{\{/;
664
665        $self->{images}->ensure($image);
666
667        my $files = $self->_prepare_step_files($step_number);
668        my $exec_env = $self->_exec_env($env, $files, defined $step->{id} ? $step->{id} : "__step_$step_number", $context->{github}{action_path});
669        my $with = $step->{with} || {};
670
671        my $rc = $self->{containers}->run_action(
672                image => $image,
673                network => $self->{network},
674                workdir => $self->_working_directory($step, $context, $status),
675                temp_path => $self->{temp_path},
676                env => $exec_env,
677                (defined $with->{entrypoint}
678                        ? (entrypoint => $self->{expression}->interpolate("$with->{entrypoint}", $context, $status))
679                        : ()),
680                arguments => $self->_action_arguments($step, $with->{args}, $context, $status),
681                title => $self->i18n('runner.explain.action', { uses => $step->{uses}, job => $job_id }),
682        );
683
684        return $self->_collected($files, $rc, 'runner.docker_action.exited');
685}

687# _action_arguments
688# Purpose:     Split a docker:// action's `with.args` into words, as a shell
689#              would.
690# Entry:       $step - for the message; $args - the text, or undef; $context;
691#              $status.
692# Exit status: Returns an array reference.  Throws runner.docker_action.bad_args
693#              when the text cannot be split (an unclosed quote), because
694#              shellwords then returns nothing and the action would run with
695#              no arguments at all.
696# Side effects: None.
697sub _action_arguments {
698        my ($self, $step, $args, $context, $status) = @_;
699
700        return [] unless defined $args;
701
702        my $text = $self->{expression}->interpolate("$args", $context, $status);
703        my @words = shellwords($text);
704
705        # Quoted as written, so a secret the interpolation filled in is not shown.
706        $self->throw('runner.docker_action.bad_args', { uses => $step->{uses}, args => "$args" })
707                if !@words && $text =~ /\S/;
708
709        return \@words;
710}

712# _run_local_action
713# Purpose:     Run an action that lives inside the project, such as
714#              "uses: ./.github/actions/test".
715# Entry:       $step; $step_number; $uses - the resolved reference; $env;
716#              $context; $status.
717# Exit status: Returns a step result hash.
718# Side effects: Reads the action's metadata; runs it.
719sub _run_local_action {
720        my ($self, $step, $step_number, $uses, $env, $context, $status) = @_;
721
722        # Resolved on the host and checked to be inside the project, links and all.
723        my ($relative, $host_dir) = @{ local_action_dir($self->{project_dir}, $uses) }{qw(relative dir)};
724        $self->throw('runner.local_action.missing', { uses => $uses, dir => $host_dir }) unless -d $host_dir;
725
726        # A metadata error is already an exception with a code of its own, and
727        # is passed on as it is.
728        my $action = App::Actions::Action->load_from_dir($host_dir);
729        my $container_path = "$WORKSPACE/$relative";
730        my $inputs = action_inputs($self->{expression}, $action, $step, $uses, $context, $status);
731
732        return $self->_run_composite_action($action, $step, $container_path, $inputs, $env, $context)
733                if $action->is_composite;
734        return $self->_run_javascript_action($action, $step, $step_number, $container_path, $inputs, $env, $context, $status)
735                if $action->is_javascript;
736
737        return $self->_unsupported_action($uses, $self->i18n('runner.unsupported.using', { using => $action->using }));
738}

740# _run_composite_action
741# Purpose:     Run a composite action's steps in a scope one level deeper,
742#              with its inputs, and an empty steps context of its own.
743# Entry:       $action; $step; $container_path - where the action is inside
744#              the container; $inputs; $env; $context.
745# Exit status: Returns a step result hash with the action's outputs.
746# Side effects: Runs the nested steps; variables they export reach the job.
747sub _run_composite_action {
748        my ($self, $action, $step, $container_path, $inputs, $env, $context) = @_;
749
750        my $uses = $step->{uses};
751        $self->throw('runner.composite.too_deep', { uses => $uses, depth => $MAX_DEPTH }) if $self->{depth} >= $MAX_DEPTH;
752
753        my %nested_ctx = (
754                %{$context},
755                inputs => $inputs,
756                steps => {},
757                job => { status => $self->{failed} ? 'failure' : 'success' },
758                github => { %{ $context->{github} || {} }, action_path => $container_path, action => $uses },
759        );
760
761        # The nested scope shares the counter, the PATH additions and the post
762        # steps (they are references), and has its own env, context and failure.
763        my $nested = bless {
764                %{$self},
765                ctx => \%nested_ctx,
766                job_env => { %{$env} },
767                depth => $self->{depth} + 1,
768                failed => 0,
769        }, ref $self;
770        $nested->_execute_step($_) for @{ $action->steps };
771
772        my %output_ctx = (%nested_ctx, env => $nested->{job_env});
773        my $declared = $action->outputs || {};
774        my %outputs = map { $_ => $self->{expression}->interpolate($declared->{$_}{value}, \%output_ctx) } keys %{$declared};
775
776        # Variables exported inside the action belong to the caller's job env.
777        for my $key (keys %{ $nested->{job_env} }) {
778                next if exists $env->{$key} && "$env->{$key}" eq "$nested->{job_env}{$key}";
779                $self->{job_env}{$key} = $nested->{job_env}{$key};
780        }
781
782        return $self->_result(
783                outcome => $nested->{failed} ? 'failure' : 'success',
784                outputs => \%outputs,
785                error => $nested->{failed} ? $self->i18n('runner.composite.failed', { uses => $uses }) : undef,
786        );
787}

789# _run_javascript_action
790# Purpose:     Run a JavaScript action with node in the job container: its
791#              pre step (under pre-if), its main step, and register its post
792#              step (under post-if) for the end of the job.
793# Entry:       $action; $step; $step_number; $container_path; $inputs; $env;
794#              $context; $status.
795# Exit status: Returns a step result hash merging what pre and main wrote.
796# Side effects: Runs node; reads the step files; may queue a post step.
797sub _run_javascript_action {
798        my ($self, $action, $step, $step_number, $container_path, $inputs, $env, $context, $status) = @_;
799
800        my $runs = $action->runs;
801        my $uses = $step->{uses};
802        $self->throw('runner.javascript.no_node', { uses => $uses }) unless $self->{has_node};
803
804        my $step_id = defined $step->{id} ? $step->{id} : "__step_$step_number";
805        my %merged = (env => {}, outputs => {}, path => [], summary => '', state => {});
806
807        my $pre = $runs->{pre};
808        if (defined $pre && length "$pre") {
809                my $condition = defined $runs->{'pre-if'} ? $runs->{'pre-if'} : 'always()';
810                my $should_run = eval { $self->{expression}->condition($condition, $context, $status) };
811                if (!defined $should_run) {
812                        $self->_complain('runner.javascript.pre_if_failed', { uses => $uses, error => trim_error($@) });
813                } elsif ($should_run) {
814                        my $rc = $self->_run_node_phase("$container_path/$pre", "$step_number-pre", $env, $step_id, $container_path, $inputs, \%merged);
815                        return $self->_result(
816                                %merged,
817                                outcome => 'failure',
818                                exit_code => $rc,
819                                error => $self->i18n('runner.javascript.pre_exited', { uses => $uses, status => $rc }),
820                        ) if $rc != 0;
821                }
822        }
823
824        my $rc = $self->_run_node_phase("$container_path/$runs->{main}", $step_number, $env, $step_id, $container_path, $inputs, \%merged);
825
826        my $post = $runs->{post};
827        push @{ $self->{post_actions} }, {
828                uses => $uses,
829                script => "$container_path/$post",
830                condition => defined $runs->{'post-if'} ? $runs->{'post-if'} : 'always()',
831                inputs => $inputs,
832                action_path => $container_path,
833                step_id => $step_id,
834                state => { %{ $merged{state} } },
835        } if defined $post && length "$post";
836
837        return $self->_result(
838                %merged,
839                outcome => $rc == 0 ? 'success' : 'failure',
840                exit_code => $rc,
841                error => $rc == 0 ? undef : $self->i18n('runner.javascript.exited', { uses => $uses, status => $rc }),
842        );
843}

845# _run_node_phase
846# Purpose:     Run one phase (pre or main) of a JavaScript action and fold
847#              what it wrote into the action's result so far.
848# Entry:       $script; $file_number - names the step files; $env; $step_id;
849#              $container_path; $inputs; $merged - the running result,
850#              including the saved state that later phases read as STATE_*.
851# Exit status: Returns node's exit status.
852# Side effects: Runs node; reads and removes the step files; updates $merged.
853sub _run_node_phase {
854        my ($self, $script, $file_number, $env, $step_id, $container_path, $inputs, $merged) = @_;
855
856        my $files = $self->_prepare_step_files($file_number);
857        my %exec_env = (%{$env}, %{ $files->{vars} });
858        $exec_env{"STATE_$_"} = $merged->{state}{$_} for keys %{ $merged->{state} };
859
860        my $rc = $self->_run_node_script($script, \%exec_env, $step_id, $container_path, $inputs);
861        my $collected = $self->_collect_step_files($files, 1);
862
863        # Later phases add to what the earlier ones wrote.
864        %{ $merged->{env} } = (%{ $merged->{env} }, %{ $collected->{env} });
865        %{ $merged->{outputs} } = (%{ $merged->{outputs} }, %{ $collected->{outputs} });
866        push @{ $merged->{path} }, @{ $collected->{path} };
867        $merged->{summary} .= $collected->{summary};
868        %{ $merged->{state} } = (%{ $merged->{state} }, %{ $collected->{state} });
869
870        return $rc;
871}

873# _run_node_script
874# Purpose:     Execute a JavaScript entry point with the job environment, its
875#              inputs as INPUT_* variables, and GITHUB_ACTION_PATH set.
876# Entry:       $script - path in the container; $env - including the step
877#              file variables; $step_id; $action_path; $inputs.
878# Exit status: Returns node's exit status.
879# Side effects: Runs docker exec.
880sub _run_node_script {
881        my ($self, $script, $env, $step_id, $action_path, $inputs) = @_;
882
883        my $exec_env = $self->_exec_env($env, undef, $step_id, $action_path);
884        # As GitHub names them, and as @actions/core's getInput reads them: upper
885        # case with spaces as _, a - or . kept (INPUT_PERL-VERSION).  Anything an
886        # environment name cannot hold becomes _ as well.
887        for my $name (keys %{$inputs}) {
888                (my $key = uc $name) =~ s/[^A-Z0-9_.-]/_/g;
889                $exec_env->{"INPUT_$key"} = $inputs->{$name};
890        }
891
892        return $self->{docker}->container_exec('-w', $WORKSPACE, @{ env_args($exec_env) }, $self->{container}, 'node', $script);
893}

895# _exec_env
896# Purpose:     The environment a command in the container runs with: the
897#              step's variables, its step files, PATH with the additions in
898#              front, and the GITHUB_ACTION values.
899# Entry:       $env; $files - from _prepare_step_files, or undef when $env
900#              already holds them; $action_id; $action_path - or undef.
901# Exit status: Returns a new hash reference.
902# Side effects: None.
903sub _exec_env {
904        my ($self, $env, $files, $action_id, $action_path) = @_;
905
906        my %exec_env = (%{$env}, $files ? %{ $files->{vars} } : ());
907        my $base = defined $exec_env{PATH} ? $exec_env{PATH} : $self->{base_path};
908        $exec_env{PATH} = join ':', @{ $self->{path_additions} }, $base;
909        $exec_env{GITHUB_ACTION} = $action_id;
910        $exec_env{GITHUB_ACTION_PATH} = defined $action_path ? $action_path : '';
911
912        return \%exec_env;
913}
